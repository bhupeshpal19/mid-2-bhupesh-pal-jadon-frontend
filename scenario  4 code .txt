  <h2>Password Strength</h2>

  <!-- Requirement 1: Password box and Show password checkbox -->
  <input type="password" id="passwordInput" placeholder="Enter password">

  <div class="checkbox-group">
    <input type="checkbox" id="showPassword">
    <label for="showPassword">Show password</label>
  </div>

  <!-- Strength bar and text label -->
  <div class="bar-container">
    <div id="strengthBar" class="strength-bar"></div>
  </div>
  <div id="strengthLabel" class="strength-label"></div>

  <!-- Checklist of five rules -->
  <ul>
    <li id="rule0">✔ At least 8 characters</li>
    <li id="rule1">✔ One uppercase letter</li>
    <li id="rule2">✔ One lowercase letter</li>
    <li id="rule3">✔ One number</li>
    <li id="rule4">✔ One special symbol (!@#$%^&* etc.)</li>
  </ul>

  <script>
    const passwordInput = document.getElementById("passwordInput");
    const showPasswordCheckbox = document.getElementById("showPassword");
    const strengthBar = document.getElementById("strengthBar");
    const strengthLabel = document.getElementById("strengthLabel");

    // Requirement 2 & Concepts: Functions in an array using regular expressions
    const ruleChecks = [
      (pwd) => pwd.length >= 8,               // At least 8 characters
      (pwd) => /[A-Z]/.test(pwd),             // One uppercase letter
      (pwd) => /[a-z]/.test(pwd),             // One lowercase letter
      (pwd) => /[0-9]/.test(pwd),             // One number
      (pwd) => /[^A-Za-z0-9]/.test(pwd)       // One special symbol
    ];

    // Requirement 5: Switch between password and text
    showPasswordCheckbox.addEventListener("change", function () {
      passwordInput.type = this.checked ? "text" : "password";
    });

    // Requirement 3 & 4: Evaluate password on every keystroke
    passwordInput.addEventListener("input", function () {
      const pwd = passwordInput.value;
      let passedCount = 0;

      // Check each rule and update checklist styling
      ruleChecks.forEach((checkFn, index) => {
        const listItem = document.getElementById(`rule${index}`);
        if (checkFn(pwd)) {
          listItem.classList.add("passed");
          passedCount++;
        } else {
          listItem.classList.remove("passed");
        }
      });

      // Update strength bar and label
      if (pwd.length === 0) {
        strengthBar.style.width = "0%";
        strengthLabel.textContent = "";
      } else if (passedCount <= 2) {
        strengthBar.style.width = "33%";
        strengthBar.style.backgroundColor = "red";
        strengthLabel.textContent = "Weak";
        strengthLabel.style.color = "red";
      } else if (passedCount <= 4) {
        strengthBar.style.width = "66%";
        strengthBar.style.backgroundColor = "orange";
        strengthLabel.textContent = "Medium";
        strengthLabel.style.color = "orange";
      } else {
        strengthBar.style.width = "100%";
        strengthBar.style.backgroundColor = "green";
        strengthLabel.textContent = "Strong";
        strengthLabel.style.color = "green";
      }
    });
  </script>
